//snapNmNum, 讀取並檢核nm.avg或nm.std(只讀取一次, 讀取拋錯視為無效; 檢測之值即使用之值):
//  undefined與null視為未給, 回傳{ ok: true, v: def }; 須為typeof為number之有限數值(不收數字字串與BigInt), min給了者須大於min; 無效回傳{ ok: false, v: def }
let snapNmNum = (nm, key, def, min) => {
    try {
        let v = nm[key]
        if (v === undefined || v === null) {
            return { ok: true, v: def }
        }
        if (typeof v !== 'number' || !Number.isFinite(v)) {
            return { ok: false, v: def }
        }
        if (min !== null && !(v > min)) {
            return { ok: false, v: def }
        }
        return { ok: true, v }
    }
    catch (err) {
        return { ok: false, v: def }
    }
}


//snapByPeriod, 讀取並檢核deal.nm.byPeriod(各鍵與各值只讀取一次, 讀取拋錯視為無效; 檢測之值即使用之值):
//  null與undefined視為未給, 回傳{ ok: true, snap: null }; 須為一般物件(可為空物件), 各值須為物件且avg為有限數值、std為有限數值且大於0(只收數值，不收數字字串與BigInt);
//  有效時snap為Map(自有且可列舉之期別標籤 => { avg, std }, 為檢核當下讀得之值), 其後正規化只用此快照, 不再讀取呼叫端之物件; 無效回傳{ ok: false, snap: null }
let snapByPeriod = (nm) => {
    try {
        let byPeriod = get(nm, 'byPeriod')
        if (byPeriod === null || byPeriod === undefined) {
            return { ok: true, snap: null }
        }
        if (!isobj(byPeriod)) {
            return { ok: false, snap: null }
        }
        let snap = new Map()
        for (let k of Object.keys(byPeriod)) {
            let v = byPeriod[k]
            if (!isobj(v)) {
                return { ok: false, snap: null }
            }
            let avg = v.avg
            let std = v.std
            if (typeof avg !== 'number' || !Number.isFinite(avg)) {
                return { ok: false, snap: null }
            }
            if (typeof std !== 'number' || !Number.isFinite(std) || std <= 0) {
                return { ok: false, snap: null }
            }
            snap.set(k, { avg, std })
        }
        return { ok: true, snap }
    }
    catch (err) {
        return { ok: false, snap: null }
    }
}


//quantile7, 已排序陣列(長度至少1, 皆為有限數值)之第7型分位數: h = (n - 1) * q，x[floor(h)] + (h - floor(h)) * (x[floor(h) + 1] - x[floor(h)])，h為整數時即x[h]
let quantile7 = (xs, q) => {
    let h = (xs.length - 1) * q
    let i = Math.floor(h)
    let f = h - i
    if (f === 0) {
        return xs[i]
    }
    return xs[i] + f * (xs[i + 1] - xs[i])
}


//statOf, 一組數值序列之統計{n,avg,std,min,max,median,q25,q75}(各期與攤平之全期皆用本函數): 只取有限數值(wsemi之arrFilterByENum; NaN、±Infinity與非數值皆不計入，與w-statistic之arrAverage、arrStd取用之母體相同)，n為有限值之個數；n為0時其餘欄皆為null
//  avg與std用w-statistic之arrAverage與arrStd，min與max用lodash(取原順序中之極值，平手取先出現者，與回傳列既有之min、max相同)，分位數為第7型(排序後之有限值，相鄰兩值皆有限，無須特例)
let statOf = (ps) => {
    let vs = arrFilterByENum(ps)
    let n = vs.length
    if (n === 0) {
        return { n, avg: null, std: null, min: null, max: null, median: null, q25: null, q75: null }
    }
    let xs = Float64Array.from(vs).sort()
    return { n, avg: arrAverage(vs), std: arrStd(vs), min: min(vs), max: max(vs), median: quantile7(xs, 0.5), q25: quantile7(xs, 0.25), q75: quantile7(xs, 0.75) }
}

 * 檢核失敗之訊息皆為固定詞彙(只含位置與期望之種類，不含輸入值)，亦不以console輸出輸入：`invalid methods[i]: must be a non-empty object`、`invalid methods[i].method: must be a non-empty string`、`invalid methods[i].method: must be a key of kp`、`invalid methods[i].deals: must be a non-empty array`、`invalid deal.keyOut: must be a non-empty string`、`invalid deal.nm: must be a non-empty object`、`invalid deal.nm.avg: must be a finite number`、`invalid deal.nm.std: must be a finite number greater than 0`、`invalid deal.nm.byPeriod: must be an object of { avg, std } with a finite avg and a finite std > 0`；nm.avg、nm.std、nm.byPeriod與其各鍵於檢核時只讀取一次(讀取拋錯視為無效)，其後正規化只用檢核時讀得之值
 * 同一次呼叫內各method之optMethod.intervalSec須相同(未給視為14400，數字字串與數值視為同值)，否則reject `invalid methods[i].optMethod.intervalSec: must be the same in every method`(i為第一個與先前不一致者之索引)，以免不同每根時長之序列靜默寫進同一資料夾；各method之optMethod.intervalSec皆未給時行為與既有相同
 * 最終計算全期攤平後序列(各期以其自己之常數正規化後)之avg、std、min、max，另增n、median、q25、q75與periods(依各期次序之陣列，各元素{period,len,n,avg,std,min,max,median,q25,q75}為該期正規化後序列之統計)，彙整為單筆結果；nm為{avg:0,std:1}且未給byPeriod時，periods即為各期原值之統計
 * 回傳列與periods之全部統計(n、avg、std、min、max、median、q25、q75)只取有限數值：NaN、±Infinity與非數值皆不計入(例如nm.std極小使正規化後溢位為±Infinity者亦不計入)，n為有限值之個數，avg與std用w-statistic之arrAverage與arrStd(取用之母體相同)，min與max為有限值中之極值，median、q25、q75為第7型分位數(排序後h=(n-1)*q，x[floor(h)]+(h-floor(h))*(x[floor(h)+1]-x[floor(h)])，n為1時即該值)；n為0者(無有限值)其餘統計欄皆為null，回傳列之min與max亦然；正規化本身不變，非有限之值仍照常逐值寫出
 *
 * @function
 * @param {String} name 輸入名稱字串
 * @param {String} symbol 輸入交易對代碼字串，例如'BTCUSDT'
 * @param {String} interval 輸入K線週期字串，例如'4hr'
 * @param {String} cointpe 輸入幣別類型字串，用於組合輸出檔名，例如'price'
 * @param {String} keyKind 輸入資料種類字串，用於組合輸出檔名，例如'ohlc'
 * @param {Array} methods 輸入指標方法設定陣列，各元素為{method,deals,optMethod}，method需存在於kp，deals為{keyIn,keyOut,nm:{avg,std,byPeriod}}之陣列(byPeriod為選配，見上)，optMethod為傳入calc函式之opt(可選，預設{})
 * @param {Array} arrOhlc 輸入K線陣列
 * @param {Object} kp 輸入method名稱對應calc函式之映射物件，函式簽章為(arr,keyIn,optMethod)=>rrs
 * @param {String} fdSelfData 輸入輸出json檔案所在資料夾路徑字串
 * @param {Object} [opt={}] 輸入設定物件，預設{}；每鍵只讀一次。funAddKeysIn與denyOutput經wsemi之getOptReader宣告(跨套件規格K3、V11)：給了即檢測，非預期之輸入(型別不符)視為未給而改用預設、不reject，opt.debug恰為true時每個鍵每個行程警告一次(固定詞彙、不含傳入之值，出口為opt.funLog或console.warn，funLog以(msg, { fn, reason, key })呼叫，fn為'w-data-trade dataConvert')；鍵之讀取拋錯視為未給。keysIn決定計算之欄位，屬K3之例外：給了而無效即reject(見opt.keysIn)
 * @param {Function} [opt.funAddKeysIn] 輸入用於逐筆轉換/擴增arrOhlc之函式，可回傳Promise，非函式視為未給，預設不轉換
 * @param {Array} [opt.keysIn=['none']] 輸入計算所用數值欄位名稱字串陣列，預設['none']；未給(undefined、null)取預設，給了須為非空陣列且元素皆為非空字串，否則reject、不改用預設且不寫出任何檔：非陣列、空陣列、鍵或元素之讀取拋錯為`invalid opt.keysIn: must be a non-empty array of non-empty strings`，元素非非空字串為`invalid opt.keysIn[i]: must be a non-empty string`(i為元素之索引；訊息皆不含輸入值)
 * @param {Boolean} [opt.denyOutput=false] 輸入是否禁止寫出json檔案布林值，只收true與false(不收1、0、'true'等，視為未給)，為true時不寫檔，預設false(不禁止)
 * @returns {Promise} 回傳Promise，resolve為各method/keyIn/keyOut組合之統計結果陣列，各元素為{method,keyIn,keyOut,avg,std,min,max,n,median,q25,q75,periods}
 * @example
 *
 * import calcMa from './calcMa.mjs'
 *
 * let arr = []

 * }
 *
 * let kp = { ma: calcMa }
 *
 * let methods = [
 *     {
 *         method: 'ma',
 *         deals: [
 *             { keyOut: 'param', nm: { avg: 0, std: 1 } },
 *         ],
 *         optMethod: {},
 *     },
 * ]
 *

 *     .then((rs) => {
 *         console.log(rs)
 *         // => [
 *         //   {
 *         //     method: 'ma',
 *         //     keyIn: 'Close',
 *         //     keyOut: 'param',
 *         //     avg: 99.968791151179,
 *         //     std: 1.3207330769565093,
 *         //     min: 95.38244060380985,
 *         //     max: 104.64005705802747,

 *         //     median: 99.98218642034857,
 *         //     q25: 99.60840606727731,
 *         //     q75: 100.35553691748575,
 *         //     periods: [ [Object], [Object], [Object], [Object], [Object], [Object] ]
 *         //   }
 *         // ]
 *         console.log(rs[0].periods.map((v) => `${v.period}:${v.len}:${v.n}`).join(' '))
 *         // => 1day:6:195 2day:12:189 4day:24:177 7day:42:159 15day:90:111 30day:180:21
 *     })
 *
 */
let dataConvert = async (name, symbol, interval, cointpe, keyKind, methods, arrOhlc, kp, fdSelfData, opt = {}) => {

    //check
    if (!isestr(name)) {
        throw new Error(`invalid name`)
    }
    if (!isestr(symbol)) {
        throw new Error(`invalid symbol`)
    }
    if (!isestr(interval)) {
        throw new Error(`invalid interval`)
    }
    if (!isestr(cointpe)) {
        throw new Error(`invalid cointpe`)
    }
    if (!isestr(keyKind)) {
        throw new Error(`invalid keyKind`)
    }
    if (!isearr(methods)) {
        throw new Error(`invalid methods`)
    }
    if (!isearr(arrOhlc)) {
        throw new Error(`invalid arrOhlc`)
    }
    if (!iseobj(kp)) {
        throw new Error(`invalid kp`)
    }
    if (!isestr(fdSelfData)) {
        throw new Error(`invalid fdSelfData`)
    }

    //opt, funAddKeysIn與denyOutput經getOptReader讀取(每鍵只讀一次; 給了即檢測, 非預期之輸入視為未給而改用預設, opt.debug為true時警告; 跨套件規格K3、V11); keysIn屬K3之例外, 見下
    let rd = getOptReader('w-data-trade dataConvert', opt)
    let funAddKeysIn = rd('funAddKeysIn', null, isfun, 'a function', 'no conversion')
    let denyOutput = rd('denyOutput', false, (v) => typeof v === 'boolean', 'a boolean', 'false')

    //keysIn, 決定計算之欄位, 屬跨套件規格K3之例外(決定資料口徑之opt): 未給(undefined、null)取預設['none']; 給了而無效即拒, 不改用預設(改用預設會得到看似成功之空輸出)
    //  鍵或元素之讀取拋錯(存取器、陷阱、已撤銷之Proxy)視為給了而無效; 鍵與元素各讀一次並取快照, 其後只用快照(檢測之值即使用之值; isearr對長度1之陣列會另讀v[0], 故不用)
    let keysIn = ['none']
    let keysInRaw
    let keysInSnap = null
    try {
        keysInRaw = ((typeof opt === 'object' && opt !== null) || typeof opt === 'function') ? opt.keysIn : undefined
        if (isarr(keysInRaw)) {
            keysInSnap = Array.from(keysInRaw)
        }
    }
    catch (err) {
        throw new Error(MSG_KEYS_IN) //讀取拋錯, 訊息固定(不含拋出之訊息)
    }

    //check, 全部檢核先於任何計算(opt.funAddKeysIn與calc)與寫檔: 任一無效即拒, 輸出夾不留任何檔
    //  訊息與先前於逐method迴圈內檢核時相同; deal.keyIn有指定而不在keysIn內之deal不會被處理, 故不檢核(與先前相同)
    //  各訊息為固定詞彙, 只含位置與期望之種類, 不含輸入值, 亦不以console輸出輸入(跨套件規格V19、V20與〈診斷輸出與遮罩〉)
    if (keysInRaw !== undefined && keysInRaw !== null) {
        if (keysInSnap === null || keysInSnap.length === 0) {
            throw new Error(MSG_KEYS_IN)
        }
        each(keysInSnap, (keyIn, i) => {
            if (!isestr(keyIn)) {
                throw new Error(`invalid opt.keysIn[${i}]: must be a non-empty string`)
            }
        })
        keysIn = keysInSnap
    }
    let intervalSecFirst = null
    let snaps = new Map() //deal => nm之快照{ avg, std, byPeriod }(檢核時讀得且已檢核之值; 每個deal物件只讀取一次)
    each(methods, (m, i) => {

        //check
        if (!iseobj(m)) {
            throw new Error(`invalid methods[${i}]: must be a non-empty object`)
        }

        //method
        let method = get(m, 'method')

        //check
        if (!isestr(method)) {
            throw new Error(`invalid methods[${i}].method: must be a non-empty string`)
        }

        //check
        if (!haskey(kp, method)) {
            throw new Error(`invalid methods[${i}].method: must be a key of kp`)
        }

        //deals
        let deals = get(m, 'deals')

        //check
        if (!isearr(deals)) {
            throw new Error(`invalid methods[${i}].deals: must be a non-empty array`)
        }

        //check, 各deal
        each(deals, (deal) => {

            //_keyIn
            let _keyIn = get(deal, 'keyIn')

            //check, 有指定keyIn且不在keysIn內者不會被處理
            if (isestr(_keyIn) && !keysIn.includes(_keyIn)) {
                return
            }

            //keyOut
            let keyOut = get(deal, 'keyOut')

            //check
            if (!isestr(keyOut)) {
                throw new Error(`invalid deal.keyOut: must be a non-empty string`)
            }

            //nm
            let nm = get(deal, 'nm')

            //check
            if (!iseobj(nm)) {
                throw new Error(`invalid deal.nm: must be a non-empty object`)
            }

            //avg、std、byPeriod, 給了即檢核(訊息固定, 不含輸入值), 並取快照供其後正規化使用(檢測之值即使用之值)
            //  avg與std為選配(undefined與null視為未給, 取avg 0、std 1), 給了須為有限數值, std須大於0(只收typeof為number者, 不收數字字串與BigInt); byPeriod見snapByPeriod
            if (!snaps.has(deal)) { //同一deal物件重複出現(例如多個method共用同一deals陣列)時, 沿用已檢核之快照, 不重讀
                let a = snapNmNum(nm, 'avg', 0, null)
                if (!a.ok) {
                    throw new Error(MSG_NM_AVG)
                }
                let s = snapNmNum(nm, 'std', 1, 0)
                if (!s.ok) {
                    throw new Error(MSG_NM_STD)
                }
                let { ok, snap } = snapByPeriod(nm)
                if (!ok) {
                    throw new Error(MSG_BY_PERIOD)
                }
                snaps.set(deal, { avg: a.v, std: s.v, byPeriod: snap })
            }

        })

        //check, optMethod.intervalSec(每根秒數): 給了而無效即拒(訊息同calc); 各method須同一每根秒數(未給視為14400, 檢測之值即使用之值), 否則不同每根時長之序列會寫進同一資料夾
        let intervalSec = readIntervalSec(get(m, 'optMethod', {}))
        if (intervalSec === null) {
            intervalSec = INTERVAL_SEC_BASE
        }
        if (intervalSecFirst === null) {
            intervalSecFirst = intervalSec
        }
        else if (intervalSecFirst !== intervalSec) {
            throw new Error(`invalid methods[${i}].optMethod.intervalSec: must be the same in every method`)
        }

    })

    //arr
    let arr = arrOhlc

    //funAddKeysIn
    if (isfun(funAddKeysIn)) {
        let ts = []
        await pmSeries(arr, async (v) => {
            let t = funAddKeysIn(v)
            if (ispm(t)) {
                t = await t
            }
            ts.push(t)
        })
        arr = ts
        // console.log('arr', arr)
    }

    //nor: 正規化 rrs 各期之 keyOut 為 param 序列, 每期呼叫 funPeriod(可選, 不傳=純轉換不落地), 回傳攤平之全期 param
    //  byPeriod(可選, null=未給): 檢核時取得之快照(Map, 期別 => { avg, std }), 某期別有自己之鍵者用其avg與std, 否則用_avg與_std
    let nor = (rrs, keyOut, _avg, _std, funPeriod, byPeriod = null) => {

        // //強制數據為未修正狀態
        // _avg = 0
        // _std = 1

        let pss = []
        each(rrs, (v) => {
            // console.log('v.vs', v.vs)

            //avgP, stdP
            let avgP = _avg
            let stdP = _std
            if (byPeriod !== null && byPeriod.has(v.period)) {
                let own = byPeriod.get(v.period)
                avgP = own.avg
                stdP = own.std
            }

            //corr
            let vs = map(v.vs, (m) => {
                let param = (m[keyOut] - avgP) / stdP
                let v = {
                    time: m.time,
                    param,
                }
                return v
            })

            //ps and push
            let ps = map(vs, 'param')
            pss.push(ps)

            //funPeriod
            if (isfun(funPeriod)) {
                funPeriod(v, vs, ps)
            }

        })

        //pss
        pss = flatten(pss)
        // console.log('pss', pss)

        return pss
    }

    //rs
    let rs = []
    await pmSeries(keysIn, async (keyIn) => {

        await pmSeries(methods, async (m) => {

            //method, deals, 各檢核已於上方先行完成(先於任何計算與寫檔)
            let method = get(m, 'method')
            let deals = get(m, 'deals')

            //optMethod: 由 opt 內 method 區塊指定 (例: opt_index_ohlc_klr 加 plusClose=0.5)
            //讓不同 cointpe + 指標可以有客製 opt (price 端可不傳, 維持 {} 預設)
            let optMethod = get(m, 'optMethod', {})

            //rrs
            let rrs = await kp[method](arr, keyIn, optMethod)
            // console.log('rrs', rrs)

            await pmSeries(deals, async(deal) => {

                //_keyIn
                let _keyIn = get(deal, 'keyIn')

                //check
                if (isestr(_keyIn)) { //若有指定keyIn才比對
                    if (keyIn !== _keyIn) {
                        return
                    }
                }

                //keyOut, 各檢核已於上方先行完成(先於任何計算與寫檔)
                let keyOut = get(deal, 'keyOut')

                //avg、std、byPeriod, 各檢核已於上方先行完成(先於任何計算與寫檔), 此處只用檢核時取得之快照(不再讀取呼叫端之物件); 未給avg與std為0與1, 未給byPeriod(null與undefined)為null
                let { avg, std, byPeriod } = snaps.get(deal)

                //funOut
                let funOut = (v, vs, ps) => {

                    //fn
                    let fn = `${cointpe}_${keyKind}_${keyIn}_${keyOut}_${method}_${v.period}.json`

                    //fp
                    let fp = path.resolve(fdSelfData, fn)

                    //fsWriteJson
                    //console.log(`writing...`, fn)
                    fsWriteJson(fp, vs)

                }
                if (denyOutput === true) {
                    funOut = () => {}
                }

                //funPeriod, 各期正規化後之統計(periods)先於寫出該期
                let periods = []
                let funPeriod = (v, vs, ps) => {
                    periods.push({ period: v.period, len: v.len, ...statOf(ps) })
                    funOut(v, vs, ps)
                }

                //pss
                let pss = nor(rrs, keyOut, avg, std, funPeriod, byPeriod)

                //r
                let r = {}
                if (true) {

                    //all, 全期攤平後之統計(只取有限數值, 與各期之統計同口徑)
                    let all = statOf(pss)
                    //console.log('keyIn', keyIn, 'method', method, 'keyOut', keyOut)
                    //console.log('avg:', all.avg, ',')
                    //console.log('std:', all.std, ',')
                    //console.log('min:', all.min, ',')
                    //console.log('max:', all.max, ',')
                    //console.log(' ')

                    //save
                    r = {
                        method,
                        keyIn,
                        keyOut,
                        avg: all.avg,
                        std: all.std,
                        min: all.min,
                        max: all.max,
                        n: all.n,
                        median: all.median,
                        q25: all.q25,
                        q75: all.q75,
                        periods,
                    }

                }

                //push
                rs.push(r)

            })

        })

    })

    return rs
}
